% blue-banner-photo — matching cover letter.
%
% Uses the CV's blue headings and orange rule for a simple prose layout.
% The renderer substitutes %%PDFMETA%%, %%HEADER%% and %%BODY%%.

\documentclass[11pt,a4paper]{article}

\usepackage[a4paper,top=1.8cm,bottom=1.8cm,left=1.9cm,right=1.9cm]{geometry}
\usepackage{fontspec}
\setmainfont{texgyreheros}[
  Extension      = .otf,
  UprightFont    = *-regular,
  BoldFont       = *-bold,
  ItalicFont     = *-italic,
  BoldItalicFont = *-bolditalic,
]
\usepackage{xcolor}
\usepackage{microtype}
\usepackage[hidelinks]{hyperref}

\definecolor{rulegray}{HTML}{F49B45}
\definecolor{accent}{HTML}{124C80}
\definecolor{muted}{gray}{0.30}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.75em}
\linespread{1.08}

% Match the CV: ragged right avoids the stretched word spacing justification
% produces at this measure. \frenchspacing removes the extra gap LaTeX puts
% after a full stop, which reads as a typo in a short letter.
\raggedright
\frenchspacing
\hyphenpenalty=10000
\exhyphenpenalty=10000
\sloppy

\newcommand{\clname}[1]{{\fontsize{20}{23}\selectfont\bfseries\color{accent} #1}\par\vspace{0.2em}}
\newcommand{\clcontact}[1]{{\small\color{muted}#1}\par\vspace{0.3em}%
  {\color{rulegray}\rule{\linewidth}{0.5pt}}\par}
\newcommand{\cldate}[1]{{\small\color{muted}#1}\par}
