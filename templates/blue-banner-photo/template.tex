% The portrait is clipped in LaTeX; the original photo remains intact.
\newcommand{\cvphoto}[1]{%
  \begin{tikzpicture}
    \clip (0,0) circle (1.16cm);
    \node[inner sep=0pt] at (0,-0.25cm) {\includegraphics[width=2.38cm]{#1}};
    \draw[white,line width=1.5pt] (0,0) circle (1.13cm);
  \end{tikzpicture}%
}
\newcommand{\cvbanner}[1]{%
  \noindent\makebox[\textwidth][c]{%
  \begin{tikzpicture}
    \node[inner xsep=0.7cm,inner ysep=0.55cm,
      left color=bannerleft,right color=bannerright] {%
      \begin{minipage}{\dimexpr\textwidth-0.3cm\relax}
        \color{white}\hypersetup{urlcolor=white,linkcolor=white}#1
      \end{minipage}%
    };
  \end{tikzpicture}}\par\vspace{0.3em}%
}
