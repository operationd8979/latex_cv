% two-column-photo — narrow sidebar with an ID photo, wide main column.
%
% Trade-off, stated plainly: a photo and a two-column layout are conventional
% in Vietnam and fine for a person to read, but they are harder on automated
% CV parsers than ats-single-column. Use this where a human reads the CV
% first; use ats-single-column where it goes through an applicant portal.
%
% The two columns are minipages, so the sidebar's text is emitted before the
% main column's. Extraction therefore reads sidebar-then-main rather than
% interleaving the two, which is what keeps it parseable at all.
%
% Compiles with Tectonic using bundled resources only: TeX Gyre Heros loaded
% by file name, no system font required.
%
% Markers: PDFMETA in the preamble, SIDEBAR and BODY inside the document.
% Macro contract: see PLACEHOLDERS.md.

\documentclass[12pt,a4paper]{article}

\usepackage[a4paper,top=1.5cm,bottom=1.5cm,left=1.2cm,right=1.2cm]{geometry}
\usepackage{fontspec}
\setmainfont{texgyreheros}[
  Extension      = .otf,
  UprightFont    = *-regular,
  BoldFont       = *-bold,
  ItalicFont     = *-italic,
  BoldItalicFont = *-bolditalic,
]
\usepackage{xcolor}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage{needspace}
\usepackage{graphicx}
\usepackage{paracol}
\usepackage{microtype}
\usepackage[normalem]{ulem}
\usepackage{hyperref}

\definecolor{rulegray}{gray}{0.60}
\definecolor{muted}{gray}{0.30}
\definecolor{accent}{HTML}{1F4E79}
\definecolor{linkblue}{HTML}{0B57D0}

% Blue and underlined, matching ats-single-column: on screen the underline is
% the only signal that a line of text is clickable, and print keeps the rule
% after it loses the colour.
\hypersetup{
  colorlinks = true,
  urlcolor   = linkblue,
  linkcolor  = linkblue,
  breaklinks = true,
}
\let\cvhref\href
\renewcommand{\href}[2]{\cvhref{#1}{\uline{#2}}}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\linespread{1.08}
\raggedright
\frenchspacing
\hyphenpenalty=10000
\exhyphenpenalty=10000
\sloppy
\clubpenalty=10000
\widowpenalty=10000
\raggedbottom

% paracol rather than two minipages: a minipage cannot break, so content one
% line too long shunts BOTH columns onto page two and leaves page one holding
% nothing but the header. paracol fills the page first and breaks normally.
\columnratio{0.315}
\setlength{\columnsep}{0.85cm}

% --- section heading --------------------------------------------------------
% \linewidth adapts to whichever minipage the heading sits in, so one macro
% serves both columns.
\titleformat{\section}
  {\normalfont\large\bfseries\color{accent}}{}{0em}{}
  [\vspace{-0.5em}{\color{rulegray}\rule{\linewidth}{0.5pt}}]
\titlespacing*{\section}{0pt}{0.95em}{0.4em}

% ===========================================================================
% Macro contract consumed by render_cv.py — see PLACEHOLDERS.md
% ===========================================================================

% Presence of this macro is what tells the renderer to stage the profile photo.
\newcommand{\cvphoto}[1]{\includegraphics[width=\linewidth]{#1}}

% Identity band: photo beside the name, running the full page width. The name
% must NOT sit in the narrow column — see the note at the top of this file.
\newcommand{\cvheaderwithphoto}[2]{%
  \begin{minipage}[c]{2.7cm}\cvphoto{#1}\end{minipage}%
  \hspace{0.75cm}%
  \begin{minipage}[c]{\dimexpr\textwidth-3.45cm\relax}#2\end{minipage}%
}
\newcommand{\cvheaderplain}[1]{#1}

\newcommand{\cvname}[1]{{\fontsize{19}{21}\selectfont\bfseries #1}\par\vspace{0.15em}}
\newcommand{\cvheadline}[1]{{\large\color{accent}#1}\par\vspace{0.35em}}
\newcommand{\cvcontact}[1]{{\small\raggedright\color{muted}#1\par}}

% Present so a plan written for a sidebar-identity template still renders.
\newcommand{\cvcontactline}[1]{{\small\raggedright\color{muted}#1\par}\vspace{0.15em}}

\newcommand{\cvsection}[1]{\Needspace{5\baselineskip}\section*{#1}}
\newcommand{\cvsummary}[1]{#1\par\vspace{0.1em}}

% \cvitem{headline}{metadata} — flush left, metadata on its own line, so PDF
% text extraction cannot detach a date from its entry.
\newcommand{\cvitem}[2]{%
  \vspace{0.4em}%
  #1\par\nopagebreak[4]
  {\small\color{muted}#2}\par\nopagebreak[3]
}
\newcommand{\cvitemplain}[1]{\vspace{0.4em}#1\par}

% \cvsubline{text} — one continuation line under an entry headline. One is
% the limit; see the note on \cvsubline in ats-single-column/template.tex.
\newcommand{\cvsubline}[1]{#1\par}

\newcommand{\cvmeta}[1]{{\small\itshape #1}\par}

% One labelled project URL per line, printed in full so a paper copy keeps a
% reachable address. The main column is narrower than a full page, so a repo
% or demo URL routinely wraps; the hanging indent holds the continuation under
% the address rather than flush with the label, where it would read as a
% separate field.
%
% Keep URLs readable and let long addresses wrap naturally.
\newcommand{\cvlinks}[1]{%
  {\small\color{muted}\hangindent=1.2em\hangafter=1 #1\par}%
}

\newlist{cvbullets}{itemize}{1}
\setlist[cvbullets]{
  leftmargin = 1.15em,
  labelsep   = 0.45em,
  itemsep    = 0.18em,
  parsep     = 0pt,
  topsep     = 0.25em,
  partopsep  = 0pt,
  label      = \textbullet,
}

% Wide-column form: label and a comma-separated run.
\newcommand{\cvskill}[2]{\vspace{0.2em}{\bfseries #1:} #2\par}

% Narrow-column form: label, then one item per line. Stacking keeps each
% keyword whole when the PDF's text is extracted.
%
% Group spacing stays comfortable even on multi-page CVs.
\newcommand{\cvskilllabel}[1]{\vspace{0.4em}{\bfseries #1}\par\vspace{0.06em}}
\newcommand{\cvskillitem}[1]{{\small #1}\par}

% Credential URLs may wrap; preserve readable type and the hanging indent.
\newcommand{\cvplain}[1]{%
  \vspace{0.15em}{\small\hangindent=0.9em\hangafter=1 #1\par}%
}
